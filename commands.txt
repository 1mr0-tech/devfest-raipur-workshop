# Shortcut used below
K="kubectl --context k3d-sre-lab"


---------------------------------------------------------------------
 0. PRE-WORK
---------------------------------------------------------------------
git clone https://github.com/1mr0-tech/devfest-raipur-workshop.git
cd devfest-raipur-workshop

npm install -g @google/gemini-cli
export GEMINI_API_KEY="PASTE_KEY_HERE"      # https://aistudio.google.com/apikey

make preflight                               # everything should say PASS

# Warm caches so the room does not all download at once
docker pull rancher/k3s:latest
docker pull nginx:1.27-alpine
docker pull busybox:1.36
npx -y kubernetes-mcp-server@latest --version


---------------------------------------------------------------------
 KEEP THE MAC AWAKE (open a spare terminal tab, leave it running)
---------------------------------------------------------------------
make awake
# or manually, 2 hours:
caffeinate -dims -t 7200
# or keep it awake only while one command runs, e.g. the agent:
caffeinate -dims gemini

# Lid closed = sleep anyway (unless on power + external display).
# Check what is holding the Mac awake:
pmset -g assertions | head -20

# After a sleep: Docker may need a moment, then
make status
k3d cluster list
k3d cluster start sre-lab                    # only if it shows 0/2 servers running


---------------------------------------------------------------------
 1. SETUP
---------------------------------------------------------------------
make up
make status
$K get nodes


---------------------------------------------------------------------
 LAB 1 - DETECT (by hand)
---------------------------------------------------------------------
make break-1

$K -n shop get pods
$K -n shop get pods -w                       # Ctrl+C to stop
$K -n shop describe pod -l app=web | tail -25
$K -n shop get events --sort-by=.lastTimestamp | tail -10

make fix-1
$K -n shop rollout status deploy/web


---------------------------------------------------------------------
 LAB 2 - DIAGNOSE (Gemini CLI, read-only)
---------------------------------------------------------------------
make agent-ro
gemini
#   First launch: choose "Trust this folder" (else MCP stays Disabled)
#   Inside gemini:  /mcp     -> kubernetes connected, 13 read-only tools
#   Inside gemini:  /quit
# Skip the trust dialog instead:
export GEMINI_CLI_TRUST_WORKSPACE=true
gemini mcp list

# Fault A again (second terminal)
make break-1

# Prompts to paste INSIDE gemini
#   Something is wrong in the shop namespace. Investigate and tell me the root cause. Do not change anything.
#   Which pods are unhealthy and why? Show the evidence (events/logs) you used.

# Fault B (harder: cause is in the logs)
make fix-1
make break-2
make status

#   The cart service keeps restarting. Find the root cause and show the log line that proves it.
#   (nudge if needed)  Check the container logs.

# Read-only agent should NOT be able to change anything
#   Scale the web deployment to 5 replicas.
#   Then run /mcp again: no write tools are listed.


---------------------------------------------------------------------
 LAB 3 - REMEDIATE (approval + dry-run)
---------------------------------------------------------------------
make agent-write
# restart gemini, then /mcp -> scale / create_or_update tools appear

#   Propose a fix for the cart crash loop. Show the exact change first. Do not apply it until I say yes.

# YOU run the dry-run before approving (nothing changes)
$K -n shop set env deploy/cart DB_HOST=postgres.shop.svc.cluster.local --dry-run=server -o jsonpath='{.spec.template.spec.containers[0].env}'

#   Yes, apply it.        (approve the tool call when prompted)

make status
$K -n shop logs -l app=cart --tail=3

# Manual equivalent if Gemini is rate-limited
make fix-2

 LAB 4 - GUARDRAILS (write profile active)
---------------------------------------------------------------------

make rbac
